cmake_minimum_required(VERSION 3.22)

if(DEFINED ENV{IDF_TARGET} AND NOT "$ENV{IDF_TARGET}" STREQUAL "esp32s31")
    message(FATAL_ERROR "Use a clean ESP32-S31 environment; IDF_TARGET must be esp32s31")
endif()
set(IDF_TARGET esp32s31 CACHE STRING "Target SoC")
if(NOT IDF_TARGET STREQUAL "esp32s31")
    message(FATAL_ERROR "This project requires esp32s31, not the source project's esp32p4")
endif()
if(NOT DEFINED ENV{IDF_PATH})
    message(FATAL_ERROR "Activate the pinned ESP-IDF 6.1.0 environment first")
endif()
include($ENV{IDF_PATH}/tools/cmake/version.cmake)
if(NOT "${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}.${IDF_VERSION_PATCH}" VERSION_EQUAL "6.1.0")
    message(FATAL_ERROR "This project is pinned to ESP-IDF 6.1.0. See docs/DEVELOPMENT.md")
endif()
set(PROJECT_VER "0.0.5-soundfont-dev")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(nektarina_s31)
